fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# Plot 1: Machine failure distribution

axes[0].set_title('Distribusi Kelas Target: Machine Failure')
axes[0].bar_label(axes[0].containers[0])

# Plot 2: Failure mode frequencies
